\section{Notation}
We write $\mathbb{N}$, $\mathbb{Z}$, $\mathbb{Q}$, and $\mathbb{R}$ for the
natural numbers, integers, rational numbers, and real numbers, respectively.
In these notes, $\mathbb{N}=\{1,2,3,\ldots\}$.

\section{A first worked example}
\begin{definition}
The derivative of a real-valued function $f$ at a point $x$ is
\[
  f'(x)=\lim_{h\to 0}\frac{f(x+h)-f(x)}{h},
\]
provided this limit exists.
\end{definition}

\begin{example}
Let $f(x)=x^2$. From the definition,
\begin{align*}
  f'(x)
    &=\lim_{h\to 0}\frac{(x+h)^2-x^2}{h}\\
    &=\lim_{h\to 0}(2x+h)\\
    &=2x.
\end{align*}
The cancellation in the second line is valid because $h\ne 0$ in the
difference quotient; the limit then describes what happens as $h$
approaches zero.
\end{example}
